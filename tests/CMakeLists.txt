function(pf_add_test name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE pf_core Qt6::Test)
    if(TARGET Qt6::Concurrent)
        target_link_libraries(${name} PRIVATE Qt6::Concurrent)
    endif()
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endfunction()

pf_add_test(tst_FinancialCalculations)
pf_add_test(tst_Database)
pf_add_test(tst_SmsParsing)
pf_add_test(tst_SmsImport)
pf_add_test(tst_Categorization)
